STEP 3 - NumPy is assignable, TF tensor is not
---------------------------------------------------------------------
a = np.ones((2, 2))
a[0, 0] = 0.0                  # works: NumPy arrays can be changed in place

t = tf.ones((2, 2))
try:
    t[0, 0] = 0.0              # fails: tf tensors are immutable
except TypeError as e:         # we catch the error to show it is expected
    print(e)

STEP 6 - GradientTape (derivative)
---------------------------------------------------------------------
x = tf.Variable(3.0)
with tf.GradientTape() as tape:     # records operations
    y = tf.square(x)                # y = x^2
dy_dx = tape.gradient(y, x)         # dy/dx = 2x = 6.0

c = tf.constant(3.0)
with tf.GradientTape() as tape:
    tape.watch(c)                   # constants must be watched manually
    y = tf.square(c)
tape.gradient(y, c)                 # 6.0

STEP 7 - Second derivative (nested tapes)
---------------------------------------------------------------------
time = tf.Variable(0.0)
with tf.GradientTape() as outer:
    with tf.GradientTape() as inner:
        position = 4.9 * time ** 2        # position = 4.9 t^2
    speed = inner.gradient(position, time)  # 1st derivative = 9.8 t
acceleration = outer.gradient(speed, time)  # 2nd derivative = 9.8

STEP 8 - Linear classifier from scratch
---------------------------------------------------------------------
8a. Make 2 classes of 2D points
negative = np.random.multivariate_normal(mean=[0, 3], cov=[[1, 0.5], [0.5, 1]], size=1000)  # class 0 cloud
positive = np.random.multivariate_normal(mean=[3, 0], cov=[[1, 0.5], [0.5, 1]], size=1000)  # class 1 cloud
inputs  = np.vstack((negative, positive)).astype(np.float32)   # stack -> shape (2000, 2)
targets = np.vstack((np.zeros((1000, 1), dtype="float32"),
                     np.ones((1000, 1), dtype="float32")))     # labels 0...0 1...1, shape (2000, 1)

8b. Weights and forward pass   (prediction = inputs x W + b)
W = tf.Variable(tf.random.uniform(shape=(2, 1)))   # 2 inputs -> 1 output
b = tf.Variable(tf.zeros(shape=(1,)))              # bias starts at 0
def model(inputs):
    return tf.matmul(inputs, W) + b                # linear model

8c. Loss (mean squared error)
def square_loss(targets, predictions):
    return tf.reduce_mean(tf.square(targets - predictions))  # average squared error

8d. One training step (gradient descent)
def training_step(inputs, targets):
    with tf.GradientTape() as tape:
        loss = square_loss(targets, model(inputs))   # forward pass + loss
    gW, gb = tape.gradient(loss, [W, b])             # gradients of loss wrt W and b
    W.assign_sub(gW * 0.1)                           # W = W - lr * gradient (lr = 0.1)
    b.assign_sub(gb * 0.1)                           # b = b - lr * gradient
    return loss

8e. Train and classify
for step in range(40):                               # 40 full-batch steps
    loss = training_step(inputs, targets)
pred_class = (model(inputs).numpy()[:, 0] > 0.5)     # output > 0.5 => class 1

8f. Decision boundary line (where prediction = 0.5)
x_line = np.linspace(-1, 4, 100)
y_line = (-W.numpy()[0, 0] / W.numpy()[1, 0]) * x_line + ((0.5 - b.numpy()[0]) / W.numpy()[1, 0])

STEP 9 - Custom Dense layer
---------------------------------------------------------------------
class SimpleDense(keras.layers.Layer):             # our own layer = subclass of Layer
    def __init__(self, units, activation=None):
        super().__init__()
        self.units = units                         # number of neurons
        self.activation = activation               # optional activation function
    def build(self, input_shape):                  # called once, when input size is known
        self.W = self.add_weight(shape=(input_shape[-1], self.units), initializer="random_normal")  # weights
        self.b = self.add_weight(shape=(self.units,), initializer="zeros")                          # bias
    def call(self, inputs):                        # forward pass
        y = tf.matmul(inputs, self.W) + self.b     # y = xW + b
        if self.activation is not None:
            y = self.activation(y)                 # apply relu etc.
        return y

layer = SimpleDense(32, activation=tf.nn.relu)
out = layer(tf.ones((2, 784)))                     # input 2x784 -> output 2x32
