=====================================================================
LAB 2 - MLP (FULLY CONNECTED NETWORK): IRIS, MNIST, FASHION-MNIST
        STEP-BY-STEP GUIDE
=====================================================================

GENERAL RECIPE FOR ANY MLP QUESTION
 1. Import libraries
 2. Load data, separate features (x) and labels (y)
 3. Preprocess  (tabular: StandardScaler | images: flatten + /255)
 4. Encode labels (one-hot OR keep integers - decides the loss)
 5. Split into train/test
 6. Layer 1 (hidden, relu)  -> Layer 2 (hidden) -> ... -> Output (softmax)
 7. Compile (optimizer + loss + metric)
 8. Train (fit)
 9. Evaluate / check overfitting / count parameters

LOSS RULE:  integer labels   -> "sparse_categorical_crossentropy"
            one-hot labels   -> "categorical_crossentropy"
PARAM RULE: Dense params = (inputs x units) + units

---------------------------------------------------------------------
STEP 1 - Libraries
---------------------------------------------------------------------
import numpy as np                            # arrays
import tensorflow as tf
from tensorflow import keras
from tensorflow.keras import layers           # Dense layers
from tensorflow.keras.models import Sequential # layer-by-layer model
from tensorflow.keras.utils import to_categorical  # one-hot encoding
from sklearn.datasets import load_iris        # iris dataset
from sklearn.preprocessing import StandardScaler  # scaling
from sklearn.model_selection import train_test_split  # split data
import matplotlib.pyplot as plt

=====================================================================
PART A - IRIS (tabular, 4 features, 3 classes)
=====================================================================
STEP 2 - Load and separate
iris = load_iris()
x = iris.data                  # shape (150, 4) -> features
y = iris.target                # shape (150,)   -> labels 0/1/2

STEP 3 - Scale features
x = StandardScaler().fit_transform(x)   # mean 0, std 1 -> helps training

STEP 4 - One-hot labels
y = to_categorical(y, 3)       # 0 -> [1,0,0], 1 -> [0,1,0], 2 -> [0,0,1]

STEP 5 - Split
x_train, x_test, y_train, y_test = train_test_split(x, y, test_size=0.2, random_state=42)  # 80/20

STEP 6 - Layers
model = Sequential([
    tf.keras.Input(shape=(4,)),            # input layer: 4 features
    layers.Dense(16, activation='relu'),   # hidden 1: 16 neurons (params = 4*16+16 = 80)
    layers.Dense(8, activation='relu'),    # hidden 2: 8 neurons  (params = 16*8+8 = 136)
    layers.Dense(3, activation='softmax')  # output: 3 class probabilities (8*3+3 = 27)
])

STEP 7 - Compile
model.compile(optimizer='adam',                  # adaptive optimizer
              loss='categorical_crossentropy',   # because labels are one-hot
              metrics=['accuracy'])

STEP 8 - Train
history = model.fit(x_train, y_train, epochs=20, batch_size=8, validation_split=0.1)
# validation_split=0.1 -> last 10% of training data is used for validation

STEP 9 - Evaluate
loss, acc = model.evaluate(x_test, y_test)   # accuracy on unseen test data

=====================================================================
PART B - MNIST (28x28 digits, 10 classes)
=====================================================================
STEP 2 - Load
from keras.datasets import mnist
(x_train, y_train), (x_test, y_test) = mnist.load_data()   # (60000,28,28) and (10000,28,28)
print(x_train.shape, y_train.shape)                        # "know your data"

STEP 3 - Flatten + normalize (MLP needs a vector, not a 2D image)
train_images = x_train.reshape((60000, 28 * 28)).astype("float32") / 255  # 784 values in 0-1
test_images  = x_test.reshape((10000, 28 * 28)).astype("float32") / 255

STEP 4 - Labels stay as integers (0-9) -> use sparse loss

STEP 6 - Layers
model = keras.Sequential([
    layers.Input(shape=(784,)),               # input: 784 pixels
    layers.Dense(512, activation='relu'),     # hidden 1: 784*512+512 = 401,920 params
    layers.Dense(256, activation='relu'),     # hidden 2: 512*256+256 = 131,328 params
    layers.Dense(10, activation='softmax')    # output: 256*10+10 = 2,570 params
])
model.summary()                               # prints layers, shapes, params

STEP 7 - Compile
model.compile(optimizer="rmsprop",
              loss="sparse_categorical_crossentropy",   # integer labels
              metrics=["accuracy"])

STEP 8 - Train
history = model.fit(train_images, y_train, epochs=10, batch_size=256, validation_split=0.1)

STEP 9 - Evaluate
model.evaluate(train_images, y_train)         # train accuracy
model.evaluate(test_images, y_test)           # test accuracy

---------------------------------------------------------------------
TASK 1 - Overfitting check
---------------------------------------------------------------------
train_loss, train_acc = model.evaluate(train_images, y_train, verbose=0)
test_loss,  test_acc  = model.evaluate(test_images,  y_test,  verbose=0)
print('Overfitting:' if train_acc - test_acc > 0.05 else 'No significant overfitting.')
# big gap (train >> test) = model memorised training data

---------------------------------------------------------------------
TASK 2 - Different hyperparameters (new model)
---------------------------------------------------------------------
train_cat = keras.utils.to_categorical(y_train, 10)   # one-hot because we change loss

model2 = keras.Sequential([
